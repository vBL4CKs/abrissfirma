func _ready() -> void:
	state = Save.read_state()
	_make_materials()
	_make_world()
	effects = Node3D.new()
	effects.process_mode = Node.PROCESS_MODE_PAUSABLE
	add_child(effects)
	player = Player.new()
	player.process_mode = Node.PROCESS_MODE_PAUSABLE
	add_child(player)
	player.action_requested.connect(use_tool)
	player.detonate_requested.connect(detonate_charges)
	player.shop_requested.connect(open_shop)
	player.pause_requested.connect(toggle_pause)
	player.tool_requested.connect(select_tool)
	player.restart_requested.connect(restart_contract)
	ui = GameUI.new()
	add_child(ui)
	ui.tool_selected.connect(select_tool)
	ui.restart_requested.connect(restart_contract)
	ui.purchase_requested.connect(purchase)
	ui.continue_requested.connect(resume_game)
	ui.contract_requested.connect(next_contract)
	ui.sensitivity_changed.connect(func(value: float): player.sensitivity = value)
	ui.quality_changed.connect(set_quality)
	ui.detonate_requested.connect(detonate_charges)
	_make_sounds()
	reset_contract()
	ui.show_briefing(job_title, reward)
	set_menu(true)
	if "--capture-preview" in OS.get_cmdline_user_args():
		call_deferred("_capture_preview")

	var environment := Environment.new()

	sky_material.sky_top_color = Color("4e8cb9")
	sky_material.sky_horizon_color = Color("ccdce0")
	sky_material.ground_horizon_color = Color("bec6b5")
	sky_material.ground_bottom_color = Color("748774")
	sky_material.sun_angle_max = 18.0
	sky.sky_material = sky_material
	environment.background_mode = Environment.BG_SKY
	environment.sky = sky
	environment.ambient_light_source = Environment.AMBIENT_SOURCE_COLOR
	environment.ambient_light_color = Color("b5cddd")
	environment.ambient_light_energy = 0.36
	environment.tonemap_mode = Environment.TONE_MAPPER_LINEAR
	environment.tonemap_exposure = 0.9
	environment.fog_enabled = true
	environment.fog_light_color = Color("b8cbd5")
	environment.fog_density = 0.0018
	environment_node.environment = environment
	add_child(environment_node)
	sun = DirectionalLight3D.new()
	sun.rotation_degrees = Vector3(-47, -34, 0)
	sun.light_color = Color("ffe3ad")
	sun.light_energy = 0.85
	sun.shadow_enabled = true
	sun.directional_shadow_max_distance = 100.0
	sun.directional_shadow_mode = DirectionalLight3D.SHADOW_PARALLEL_4_SPLITS
	add_child(sun)
	static_box(self, Vector3(0, -0.3, 0), Vector3(230, 0.6, 230), materials["grass"])
	static_box(self, Vector3(0, -0.08, 0), Vector3(45, 0.18, 43), materials["dirt"])
	static_box(self, Vector3(0, 0.01, 0), Vector3(12.5, 0.14, 10.1), materials["concrete"])
	var art := SiteArt.new()
	add_child(art)
	art.build_site()

func _make_materials() -> void:
	var colors := {"brick": "c28365", "cream": "dfcda8", "sage": "9ab6a5", "blue": "8fb0b9", "rose": "c79d8c", "concrete": "aeb4ad", "wood": "a47b52", "darkwood": "6c5543", "steel": "71818a", "glass": "80c5d0", "dirt": "bcad90", "grass": "82976b", "yellow": "eeba42", "ink": "263e4e", "white": "eee5d1"}
	for key in colors:
		var material := StandardMaterial3D.new()
		material.albedo_color = Color(colors[key])
		material.roughness = 0.9
		if key == "steel":
			material.metallic = 0.65
			material.roughness = 0.45
		if key == "glass":
			material.roughness = 0.2
			material.metallic = 0.35
		materials[key] = material
	var image := Image.create(128, 128, false, Image.FORMAT_RGB8)
	var rng := RandomNumberGenerator.new()
	rng.seed = 204
	for y in range(128):
		for x in range(128):
			var offset := 32 if (y / 32) % 2 else 0
			var mortar := y % 32 < 2 or (x + offset) % 64 < 2
			var value := rng.randf_range(0.72, 0.78) if mortar else rng.randf_range(0.94, 1.0)
			image.set_pixel(x, y, Color(value, value, value))
	var texture := ImageTexture.create_from_image(image)
	for key in ["brick", "cream", "sage", "rose", "blue"]:
		var material: StandardMaterial3D = materials[key]
		material.albedo_texture = texture
		material.uv1_triplanar = true
		material.uv1_world_triplanar = true
		material.uv1_scale = Vector3.ONE * 0.8

func reset_contract() -> void:
	job_generation += 1
	support_pending = false
	completion_pending = false
	if is_instance_valid(building):
		remove_child(building)
		building.queue_free()
	for item in debris:
		if is_instance_valid(item):
			item.queue_free()
	debris.clear()
	for item in charges:
		if is_instance_valid(item["node"]):
			item["node"].queue_free()
	charges.clear()
	for child in effects.get_children():
		child.queue_free()
	pieces.clear()
	stair_ids.clear()
	stair_ramp = null
	removed_count = 0
	paid = false
	supply = [0, 12, 6]
	reward = 1200 + mini(int(state.completed), 8) * 100
	var names := ["Haus Sonneneck", "Villa Morgenrot", "Haus am Kran"]
	job_title = "%02d · %s" % [int(state.completed) + 1, names[int(state.completed) % names.size()]]
	building = Node3D.new()
	building.process_mode = Node.PROCESS_MODE_PAUSABLE
	building.name = "ZerstoerbaresHaus"
	add_child(building)
	_build_house()
	_connect_structure()
	player.position = Vector3(13, 0.25, 17)
	player.velocity = Vector3.ZERO
	player.rotation = Vector3.ZERO
	player.head.rotation = Vector3.ZERO
	player.look_at(Vector3(0, 0.25, 0), Vector3.UP)
	player.head.rotation.x = -0.05
	apply_upgrades()
	select_tool(0)
	update_hud()

func _build_house() -> void:
	var palette := ["cream", "sage", "rose", "blue"]
	var shift := int(state.completed) % 4
	for level in range(2):
		var base_y := float(level) * 3.7
		for row in range(3):
			var y := base_y + 0.65 + float(row) * 1.2
			for column in range(10):
				var x := -5.4 + float(column) * 1.2
				for face in [-1, 1]:
					if face == 1 and level == 0 and column in [4, 5] and row < 2:
						continue
					if row == 1 and column in [1, 2, 7, 8]:
						continue
					var mat_name: String = palette[(level * 2 + (1 if face > 0 else 0) + shift) % 4]
					add_piece(Vector3(x, y, face * 4.8), Vector3(1.18, 1.18, 0.28), "brick", mat_name)
			for column in range(8):
				var z := -4.2 + float(column) * 1.2
				for face in [-1, 1]:
					if row == 1 and column in [2, 5]:
						continue
					add_piece(Vector3(face * 6.0, y, z), Vector3(0.28, 1.18, 1.18), "brick", palette[(level + shift) % 4])
		# A lightweight interior partition leaves a wide doorway through both rooms.
		for z in [-3.6, -2.4, 2.4, 3.6]:
			for row in range(3):
				add_piece(Vector3(0, base_y + 0.65 + row * 1.2, z), Vector3(0.16, 1.18, 1.18), "wood", "sage" if level == 0 else "rose")
		# Individual floor plates participate in the support network.
		var floor_y := 3.7 * float(level + 1)
		for x in range(10):
			for z in range(8):
				# Opening over the stairs gives real access to the upper floor.
				if level == 0 and x >= 7 and z >= 1:
					continue
				add_piece(Vector3(-5.4 + x * 1.2, floor_y, -4.2 + z * 1.2), Vector3(1.19, 0.23, 1.19), "concrete", "concrete")
	# Glass and timber frames are destructible too.
	for level in range(2):
		for x in [-3.6, 3.6]:
			for z in [-4.82, 4.82]:
				var y := level * 3.7 + 1.85
				add_piece(Vector3(x, y, z), Vector3(2.35, 1.12, 0.07), "glass", "glass")
				add_piece(Vector3(x, y - 0.61, z), Vector3(2.5, 0.11, 0.38), "wood", "white")
				add_piece(Vector3(x, y + 0.61, z), Vector3(2.5, 0.1, 0.35), "wood", "white")
				add_piece(Vector3(x, y, z + 0.01), Vector3(0.07, 1.2, 0.12), "wood", "white")
	# Destructible stairs: low steps are ground-anchored, the rest connects upward.
	for step in range(18):
		stair_ids.append(pieces.size())
		add_piece(Vector3(4.45, 0.106 + step * 0.212, 4.1 - step * 0.44), Vector3(2.0, 0.212, 0.46), "wood", "wood")
	# Player-only ramp smooths the small steps; tools still hit destructible treads.
	stair_ramp = StaticBody3D.new()
	stair_ramp.name = "BegehbareTreppenrampe"
	stair_ramp.collision_layer = 16
	stair_ramp.collision_mask = 0
	stair_ramp.position = Vector3(4.45, 0, 0)
	var ramp_shape := CollisionShape3D.new()
	var ramp_wedge := ConvexPolygonShape3D.new()
	ramp_wedge.points = PackedVector3Array([
		Vector3(-1, -0.12, 4.85), Vector3(1, -0.12, 4.85),
		Vector3(-1, 0, 4.85), Vector3(1, 0, 4.85),
		Vector3(-1, -0.12, -3.65), Vector3(1, -0.12, -3.65),
		Vector3(-1, 4.06, -3.65), Vector3(1, 4.06, -3.65)])
	ramp_shape.shape = ramp_wedge
	stair_ramp.add_child(ramp_shape)
	building.add_child(stair_ramp)
	# Furnishings are intentionally simple, built from reusable breakable pieces.
	_table(Vector3(-3.3, 0, 0.2))
	_table(Vector3(-3.3, 3.83, -1.0))
	add_piece(Vector3(-4.9, 0.45, -3.8), Vector3(1.4, 0.9, 0.65), "wood", "white")
	add_piece(Vector3(-3.8, 4.1, 2.6), Vector3(2.3, 0.5, 1.3), "wood", "blue")
	# A house number and company plaque are attached to walls and fall with them.
	var number := Label3D.new()
	number.text = "17"
	number.font_size = 80
	number.pixel_size = 0.005
	var number_wall: AbrissPiece = pieces[0]
	for candidate in pieces:
		if candidate.position.distance_to(Vector3(1.8, 3.05, 4.8)) < 0.2:
			number_wall = candidate
			break
	number_wall.add_child(number)
	number.global_position = Vector3(1.9, 2.7, 4.96)

func _table(at: Vector3) -> void:
	add_piece(at + Vector3(0, 0.95, 0), Vector3(2.2, 0.16, 1.2), "wood", "wood")
	for x in [-0.85, 0.85]:
		for z in [-0.4, 0.4]:
			add_piece(at + Vector3(x, 0.45, z), Vector3(0.12, 0.9, 0.12), "wood", "darkwood")

func add_piece(at: Vector3, size: Vector3, kind: String, material_name: String) -> void:
	var piece := Piece.new()
	piece.setup(pieces.size(), at, size, kind, materials[material_name])
	building.add_child(piece)
	pieces.append(piece)

func _connect_structure() -> void:
	for i in range(pieces.size()):
		var a := pieces[i]
		for j in range(i + 1, pieces.size()):
			var b := pieces[j]
			var difference := (a.position - b.position).abs()
			var reach := (a.size + b.size) * 0.5 + Vector3.ONE * 0.095
			if difference.x <= reach.x and difference.y <= reach.y and difference.z <= reach.z:
				a.neighbors.append(j)
				b.neighbors.append(i)

func use_tool() -> void:
	if get_tree().paused:
		return
	var hit := player.aim(3.6 if selected_tool == 0 else 8.0)
	if hit.is_empty():
		ui.show_toast("Geh näher an das Haus und ziele auf ein Bauteil.")
		return
	var piece := hit["collider"] as AbrissPiece
	if piece == null or piece.removed:
		ui.show_toast("Dieses Objekt gehört zur Baustellenkulisse. Ziele auf das Abrisshaus.")
		return
	player.swing()
	if selected_tool == 0:
		var force := 82.0 if state.owned.get("hammer_upgrade", false) else 38.0
		var point: Vector3 = hit["position"]
		if piece.hit(force):
			remove_piece(piece, point, 2.0)
			request_support_check()
		_spawn_dust(point, piece.material.albedo_color, 5, 0.5)
		_play_sound("glass" if piece.kind == "glass" else "hit", point)
		player.shake = 0.12
	else:
		place_charge(piece, hit["position"], hit["normal"], selected_tool)
	update_hud()

func place_charge(piece: AbrissPiece, point: Vector3, normal: Vector3, type: int) -> void:
	if supply[type] <= 0:
		ui.show_toast("Vorrat aufgebraucht. Der Hammer bleibt kostenlos verfügbar.")
		return
	for charge in charges:
		if charge["node"].global_position.distance_to(point) < 0.6:
			ui.show_toast("Hier liegt bereits eine Ladung. Nutze ein anderes Bauteil.")
			return
	var gadget := Node3D.new()
	piece.add_child(gadget)
	gadget.global_position = point + normal * 0.13
	var color := Color("ff933d") if type == 1 else Color("a590ff")
	var material := StandardMaterial3D.new()
	material.albedo_color = color
	material.emission_enabled = true
	material.emission = color
	material.emission_energy_multiplier = 1.2
	visual_box(gadget, Vector3.ZERO, Vector3(0.26, 0.26, 0.26), materials["ink"])
	visual_box(gadget, Vector3(0, 0, 0.14), Vector3(0.18, 0.18, 0.04), material)
	var light := OmniLight3D.new()
	light.light_color = color
	light.light_energy = 0.6
	light.omni_range = 1.0
	gadget.add_child(light)
	charges.append({"node": gadget, "type": type, "piece": piece})
	supply[type] -= 1
	ui.show_toast("%s platziert · R zum Zünden · Noch %d im Vorrat" % ["Impulsladung" if type == 1 else "Wirbelladung", supply[type]])
	_play_sound("place", point)

func detonate_charges() -> void:
	if get_tree().paused:
		return
	if charges.is_empty():
		ui.show_toast("Erst mit 2 oder 3 ein Gerät wählen und mit Linksklick platzieren.")
		return
	var pending := charges.duplicate()
	charges.clear()
	for item in pending:
		var gadget: Node3D = item["node"]
		if not is_instance_valid(gadget):
			continue
		var point := gadget.global_position
		gadget.queue_free()
		blast(point, int(item["type"]))
	request_support_check()
	update_hud()

func blast(point: Vector3, type: int) -> void:
	var enhanced: bool = state.owned.get("blast_upgrade", false)
	var radius := (4.2 if type == 1 else 5.4) * (1.2 if enhanced else 1.0)
	var damage := (160.0 if type == 1 else 95.0) * (1.35 if enhanced else 1.0)
	for piece in pieces:
		if piece.removed:
			continue
		var distance := piece.global_position.distance_to(point)
		if distance < radius and piece.hit(damage * (1.0 - distance / (radius * 1.4))):
			remove_piece(piece, point, 9.0 if type == 1 else -6.0)
	_spawn_dust(point, Color("c5b49a"), 24 if quality else 9, radius * 0.4)
	var ring := MeshInstance3D.new()
	var sphere := SphereMesh.new()
	sphere.radius = 0.5
	sphere.height = 1.0
	ring.mesh = sphere
	ring.position = point
	var material := StandardMaterial3D.new()
	material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	material.albedo_color = Color(1.0, 0.67, 0.22, 0.4) if type == 1 else Color(0.6, 0.4, 1.0, 0.35)
	ring.material_override = material
	effects.add_child(ring)
	var tween := create_tween().set_parallel()
	tween.tween_property(ring, "scale", Vector3.ONE * radius * 1.7, 0.38)
	tween.tween_property(material, "albedo_color:a", 0.0, 0.4)
	tween.chain().tween_callback(ring.queue_free)
	player.shake = minf(0.8, radius / maxf(player.position.distance_to(point), 1.0))
	_play_sound("blast", point)

func remove_piece(piece: AbrissPiece, origin: Vector3, force: float) -> void:
	if piece.removed:
		return
	# Gadgets become independent so an earlier hit never silently deletes a placed charge.
	for charge in charges:
		if charge["piece"] == piece and is_instance_valid(charge["node"]):
			charge["node"].reparent(effects)
	# Pending charges are still children during a simultaneous detonation; pieces persist hidden.
	var center := piece.global_position
	piece.deactivate()
	if piece.piece_id in stair_ids and is_instance_valid(stair_ramp):
		stair_ramp.collision_layer = 0
	removed_count += 1
	var body := RigidBody3D.new()
	body.mass = clampf(piece.size.x * piece.size.y * piece.size.z * 4.0, 0.5, 12.0)
	body.collision_layer = 8
	body.collision_mask = 1 | 2
	body.position = center
	body.linear_damp = 0.5
	body.angular_damp = 0.7
	var box := BoxShape3D.new()
	box.size = piece.size * 0.91
	var collision := CollisionShape3D.new()
	collision.shape = box
	body.add_child(collision)
	var mesh := visual_box(body, Vector3.ZERO, piece.size * 0.94, piece.material)
	mesh.rotation = Vector3(randf_range(-0.08, 0.08), randf_range(-0.1, 0.1), randf_range(-0.08, 0.08))
	effects.add_child(body)
	var away := (center - origin).normalized()
	if away.length_squared() < 0.1:
		away = Vector3.UP
	body.apply_central_impulse((away * force + Vector3.UP * absf(force) * 0.25) * body.mass)
	body.apply_torque_impulse(Vector3(randf_range(-1, 1), randf_range(-1, 1), randf_range(-1, 1)) * 2.0)
	debris.append(body)
	while debris.size() > MAX_DEBRIS:
		var oldest := debris.pop_front() as RigidBody3D
		if is_instance_valid(oldest):
			oldest.queue_free()
	var generation := job_generation
	get_tree().create_timer(7.0, false).timeout.connect(func():
		if is_instance_valid(body) and generation == job_generation:
			body.freeze = true
			body.collision_layer = 0
			body.collision_mask = 0
	)

func request_support_check() -> void:
	if support_pending:
		return
	support_pending = true
	var generation := job_generation
	get_tree().create_timer(0.12, false).timeout.connect(func():
		if generation != job_generation:
			return
		support_pending = false
		check_support()
	)

func check_support() -> int:
	var supported: Dictionary = {}
	var queue: Array[int] = []
	for piece in pieces:
		if not piece.removed and piece.anchored:
			supported[piece.piece_id] = true
			queue.append(piece.piece_id)
	var cursor := 0
	while cursor < queue.size():
		var current := pieces[queue[cursor]]
		cursor += 1
		for neighbor in current.neighbors:
			if not pieces[neighbor].removed and not supported.has(neighbor):
				supported[neighbor] = true
				queue.append(neighbor)
	var fallen := 0
	for piece in pieces:
		if not piece.removed and not supported.has(piece.piece_id):
			remove_piece(piece, piece.global_position - Vector3.UP, 0.3)
			fallen += 1
	if fallen > 8:
		_spawn_dust(Vector3(0, 2.4, 0), Color("c5b49a"), 18 if quality else 7, 4.0)
		_play_sound("blast", Vector3(0, 2, 0))
		ui.show_toast("Einsturz! %d Bauteile haben ihre Stütze verloren." % fallen)
	update_hud()
	check_completion()
	return fallen

func check_completion() -> void:
	if paid or completion_pending or progress() < TARGET:
		return
	completion_pending = true
	var generation := job_generation
	get_tree().create_timer(2.6, false).timeout.connect(func():
		if generation == job_generation and not paid:
			complete_contract()
	)

func complete_contract() -> void:
	if paid or progress() < TARGET:
		return
	paid = true
	completion_pending = false
	var earned := reward + (supply[1] + supply[2]) * 10
	state.money = int(state.money) + earned
	state.completed = int(state.completed) + 1
	save_progress()
	update_hud()
	ui.show_result(earned, int(state.money))
	set_menu(true)

func purchase(id: String) -> void:
	if not COSTS.has(id) or state.owned.get(id, false):
		return
	var cost: int = COSTS[id]
	if int(state.money) < cost:
		ui.show_toast("Dafür reicht dein Guthaben noch nicht.")
		return
	state.money = int(state.money) - cost
	state.owned[id] = true
	apply_upgrades()
	save_progress()
	update_hud()
	ui.show_shop(int(state.money), state.owned)
	ui.show_toast("Verbesserung gekauft und gespeichert.")

func apply_upgrades() -> void:
	player.speed = 8.0 if state.owned.get("boots_upgrade", false) else 6.2

func select_tool(index: int) -> void:
	selected_tool = clampi(index, 0, 2)
	player.select_tool(selected_tool)
	update_hud()

func update_hud() -> void:
	if not is_instance_valid(ui):
		return
	ui.set_status(int(state.money), progress(), selected_tool, charges.size(), job_title)
	if selected_tool > 0:
		ui.charges_label.text = "%d platziert · %d im Vorrat · Ziel: 72 %% Abriss" % [charges.size(), supply[selected_tool]]
	else:
		ui.charges_label.text = "%d Ladungen platziert · Ziel: 72 %% Abriss" % charges.size()

func progress() -> float:
	return float(removed_count) / maxf(float(pieces.size()), 1.0)

func save_progress() -> void:
	if not Save.write_state(state):
		ui.show_toast("Spielstand konnte nicht geschrieben werden. Bitte Ordnerrechte prüfen.")

func set_menu(open: bool) -> void:
	get_tree().paused = open
	player.controls_enabled = not open
	Input.mouse_mode = Input.MOUSE_MODE_VISIBLE if open else Input.MOUSE_MODE_CAPTURED

func resume_game() -> void:
	briefing = false
	ui.hide_panels()
	set_menu(false)

func toggle_pause() -> void:
	if ui.is_panel_open():
		resume_game()
	else:
		ui.show_pause()
		set_menu(true)

func open_shop() -> void:
	if ui.is_panel_open():
		return
	ui.show_shop(int(state.money), state.owned)
	set_menu(true)

func restart_contract() -> void:
	if paid:
		ui.show_toast("Der Auftrag ist bezahlt. Nimm im Shop den nächsten Auftrag an.")
		return
	reset_contract()
	resume_game()
	ui.show_toast("Auftrag neu gestartet · Werkzeugvorrat aufgefüllt.")

func next_contract() -> void:
	if briefing:
		resume_game()
		ui.show_toast("Willkommen! Erkunde das Haus. 1 Hammer · 2 Impuls · 3 Wirbel · R Zünden")
	elif paid:
		reset_contract()
		resume_game()
		ui.show_toast("Neuer Auftrag · Neues Werkzeugpaket · Viel Erfolg!")
	else:
		ui.show_toast("Schließe zuerst den laufenden Auftrag ab: 72 % Abriss.")

func set_quality(enabled: bool) -> void:
	quality = enabled
	sun.shadow_enabled = enabled
	player.camera.far = 260.0 if enabled else 150.0

func _unhandled_input(event: InputEvent) -> void:
	# This node processes while paused so Escape can also close menus.
	if event is InputEventKey and event.pressed and not event.echo:
		if event.physical_keycode == KEY_ESCAPE:
			toggle_pause()
			get_viewport().set_input_as_handled()

func _enter_tree() -> void:
	process_mode = Node.PROCESS_MODE_ALWAYS

func _spawn_dust(at: Vector3, color: Color, count: int, spread: float) -> void:
	var particles := CPUParticles3D.new()
	particles.amount = count
	particles.one_shot = true
	particles.lifetime = 1.4
	particles.explosiveness = 0.92
	particles.emission_shape = CPUParticles3D.EMISSION_SHAPE_SPHERE
	particles.emission_sphere_radius = spread
	particles.direction = Vector3.UP
	particles.spread = 100.0
	particles.initial_velocity_min = 0.5
	particles.initial_velocity_max = 2.3
	particles.gravity = Vector3(0, -0.6, 0)
	particles.scale_amount_min = 0.15
	particles.scale_amount_max = 0.65
	var gradient := Gradient.new()
	gradient.set_color(0, Color(color, 0.55))
	gradient.set_color(1, Color(color, 0.0))
	particles.color_ramp = gradient
	var mesh := SphereMesh.new()
	mesh.radius = 0.5
	mesh.height = 1.0
	mesh.radial_segments = 6
	mesh.rings = 3
	var material := StandardMaterial3D.new()
	material.albedo_color = Color(color, 0.5)
	material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	material.vertex_color_use_as_albedo = true
	material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	mesh.material = material
	particles.mesh = mesh
	particles.position = at
	effects.add_child(particles)
	particles.emitting = true
	particles.finished.connect(particles.queue_free)

func _make_sounds() -> void:
	for type in ["hit", "blast", "glass", "place"]:
		var duration := 0.18 if type != "blast" else 1.05
		var samples := PackedByteArray()
		var rate := 22050
		var rng := RandomNumberGenerator.new()
		rng.seed = 71
		for i in range(int(duration * rate)):
			var t := float(i) / rate
			var envelope := exp(-t * (5.0 if type == "blast" else 22.0))
			var frequency := 62.0 if type == "blast" else (160.0 if type == "hit" else 930.0)
			var wave := (sin(t * TAU * frequency) * 0.55 + rng.randf_range(-1, 1) * 0.45) * envelope
			var value := int(clampf(wave * 21000, -32767, 32767))
			samples.append(value & 255)
			samples.append((value >> 8) & 255)
		var audio := AudioStreamWAV.new()
		audio.format = AudioStreamWAV.FORMAT_16_BITS
		audio.mix_rate = rate
		audio.data = samples
		sounds[type] = audio

func _play_sound(type: String, at: Vector3) -> void:
	if muted:
		return
	var sound := AudioStreamPlayer3D.new()
	sound.stream = sounds[type]
	sound.position = at
	sound.max_distance = 65.0
	sound.volume_db = -7.0 if type == "blast" else -12.0
	sound.pitch_scale = randf_range(0.9, 1.1)
	effects.add_child(sound)
	sound.play()
	sound.finished.connect(sound.queue_free)

func visual_box(parent: Node3D, at: Vector3, size: Vector3, material: Material) -> MeshInstance3D:
	var visual := MeshInstance3D.new()
	var mesh := BoxMesh.new()
	mesh.size = size
	visual.mesh = mesh
	visual.position = at
	visual.material_override = material
	parent.add_child(visual)
	return visual

func static_box(parent: Node3D, at: Vector3, size: Vector3, material: Material) -> void:
	var body := StaticBody3D.new()
	body.position = at
	parent.add_child(body)
	visual_box(body, Vector3.ZERO, size, material)
	var shape := CollisionShape3D.new()
	var box := BoxShape3D.new()
	box.size = size
	shape.shape = box
	body.add_child(shape)
